local Connections = {}
local RefreshCallbacks = {}

local function trackConn(conn)
    table.insert(Connections, conn)
    return conn
end

local function clearConnections()
    for _, conn in ipairs(Connections) do
        if conn and conn.Disconnect then
            conn:Disconnect()
        end
    end
    table.clear(Connections)
end

local function addCorner(inst, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius or 10)
    c.Parent = inst
    return c
end

local function addStroke(inst, color, thick, transp)
    local s = Instance.new("UIStroke")
    s.Color = color or Theme.BorderPurple
    s.Thickness = thick or 1.2
    s.Transparency = transp or 0.25
    s.Parent = inst
    return s
end

local function addGradient(inst, c1, c2, rot)
    local g = Instance.new("UIGradient")
    g.Color = ColorSequence.new(c1, c2)
    g.Rotation = rot or 45
    g.Parent = inst
    return g
end

local function makeDraggable(handle, target, onTap)
    local dragging = false
    local moved = false
    local activeInput = nil
    local dragStart = Vector3.zero
    local startPos = target.Position

    trackConn(handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            moved = false
            activeInput = input
            dragStart = input.Position
            startPos = target.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    if activeInput == input then
                        dragging = false
                        activeInput = nil
                        if not moved and onTap then
                            onTap()
                        end
                    end
                end
            end)
        end
    end))

    trackConn(UserInputService.InputChanged:Connect(function(input)
        if dragging and input == activeInput then
            local delta = input.Position - dragStart
            if delta.Magnitude > 6 then
                moved = true
            end
            local vp = Workspace.CurrentCamera.ViewportSize
            local maxX = math.max(6, vp.X - target.AbsoluteSize.X - 6)
            local maxY = math.max(6, vp.Y - target.AbsoluteSize.Y - 6)
            local nx = math.clamp(startPos.X.Offset + delta.X, 6, maxX)
            local ny = math.clamp(startPos.Y.Offset + delta.Y, 6, maxY)
            target.Position = UDim2.new(0, nx, 0, ny)
        end
    end))
end

local function createPage(id)
    local sf = Instance.new("ScrollingFrame")
    sf.Name = id .. "Page"
    sf.Size = UDim2.new(1, 0, 1, 0)
    sf.BackgroundTransparency = 1
    sf.ScrollBarThickness = 4
    sf.ScrollBarImageColor3 = Theme.PurpleBright
    sf.CanvasSize = UDim2.new(0, 0, 0, 0)
    sf.AutomaticCanvasSize = Enum.AutomaticSize.Y
    sf.Visible = false
    sf.Parent = ContentFrame

    local ll = Instance.new("UIListLayout")
    ll.Padding = UDim.new(0, 7)
    ll.Parent = sf

    local pad = Instance.new("UIPadding")
    pad.PaddingRight = UDim.new(0, 6)
    pad.PaddingBottom = UDim.new(0, 8)
    pad.Parent = sf

    Pages[id] = sf
    return sf
end

local function selectTab(id)
    for k, p in pairs(Pages) do
        p.Visible = (k == id)
    end
    for k, b in pairs(TabBtns) do
        local active = (k == id)
        b.BackgroundColor3 = active and Theme.PurplePrimary or Theme.BgDark
        b.TextColor3 = active and Theme.TextMain or Theme.TextMuted
    end
end

local function addTab(id, titleKey)
    createPage(id)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, 0, 0, 34)
    btn.BackgroundColor3 = Theme.BgDark
    btn.Text = T(titleKey)
    btn.TextColor3 = Theme.TextMuted
    btn.Font = Enum.Font.GothamSemibold
    btn.TextSize = 11
    btn.Parent = Sidebar
    addCorner(btn, 8)

    btn.MouseButton1Click:Connect(function()
        selectTab(id)
    end)

    table.insert(RefreshCallbacks, function()
        btn.Text = T(titleKey)
    end)

    TabBtns[id] = btn
end

local function makeSlider(parent, labelKey, minV, maxV, step, getVal, setVal)
    local card = Instance.new("Frame")
    card.Size = UDim2.new(1, 0, 0, 46)
    card.BackgroundColor3 = Theme.BgCard
    card.Parent = parent
    addCorner(card, 8)
    addStroke(card, Theme.PurpleDark, 1, 0.4)

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(1, -16, 0, 18)
    lbl.Position = UDim2.new(0, 10, 0, 4)
    lbl.BackgroundTransparency = 1
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.TextColor3 = Theme.TextMain
    lbl.Font = Enum.Font.Gotham
    lbl.TextSize = 11
    lbl.Parent = card

    local bar = Instance.new("Frame")
    bar.Size = UDim2.new(1, -20, 0, 8)
    bar.Position = UDim2.new(0, 10, 0, 29)
    bar.BackgroundColor3 = Color3.fromRGB(40, 28, 62)
    bar.Parent = card
    addCorner(bar, 999)

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new(0, 0, 1, 0)
    fill.BackgroundColor3 = Theme.PurpleBright
    fill.Parent = bar
    addCorner(fill, 999)

    local function syncSlider()
        local v = getVal()
        lbl.Text = string.format("%s: %s", T(labelKey), tostring(v))
        local ratio = math.clamp((v - minV) / math.max(maxV - minV, 1e-6), 0, 1)
        fill.Size = UDim2.new(ratio, 0, 1, 0)
    end
    syncSlider()

    local dragging = false
    local function applyX(xPos)
        local rel = math.clamp((xPos - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
        local raw = minV + (maxV - minV) * rel
        local snapped = math.floor((raw / step) + 0.5) * step
        snapped = math.clamp(snapped, minV, maxV)
        setVal(snapped)
        syncSlider()
    end

    trackConn(bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            applyX(input.Position.X)
        end
    end))

    trackConn(UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            applyX(input.Position.X)
        end
    end))

    trackConn(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end))

    table.insert(RefreshCallbacks, syncSlider)
end

local function findNearestCombatTarget()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end

    local bestPart, bestDist = nil, math.huge
    for _, model in ipairs(Workspace:GetDescendants()) do
        if model:IsA("Model") and model ~= char then
            local hum = model:FindFirstChildOfClass("Humanoid")
            local hrp = model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart
            if hum and hrp and hum.Health > 0 and not Players:GetPlayerFromCharacter(model) then
                local d = (hrp.Position - root.Position).Magnitude
                if d < bestDist then
                    bestDist = d
                    bestPart = hrp
                end
            end
        end
    end
    return bestPart, bestDist
end

    local target, dist = findNearestCombatTarget()
    if target then

    else
        combatStatus.Text = "No active NPC/Enemy model found nearby."
    end
end)

local function scanUgcTargets()
    local results = {}
    local actionParts = {}

    for _, obj in ipairs(Workspace:GetDescendants()) do
        local nameLower = string.lower(obj.Name)
        local isUgcWord = string.find(nameLower, "ugc") or string.find(nameLower, "limited") or string.find(nameLower, "claim") or string.find(nameLower, "free")

        if obj:IsA("ProximityPrompt") then
            local parentName = obj.Parent and string.lower(obj.Parent.Name) or ""
            local actLower = string.lower(obj.ActionText or "")
            local objTextLower = string.lower(obj.ObjectText or "")
            if isUgcWord or string.find(parentName, "ugc") or string.find(parentName, "event") or string.find(actLower, "ugc") or string.find(objTextLower, "ugc") or string.find(actLower, "claim") then
                local part = obj.Parent and (obj.Parent:IsA("BasePart") and obj.Parent or obj.Parent:FindFirstChildWhichIsA("BasePart"))
                if part then
                    table.insert(actionParts, {part = part, prompt = obj})
                end
                if #results < 5 then
                    table.insert(results, obj.Parent and obj.Parent.Name or obj.Name)
                end
            end
        elseif obj:IsA("BasePart") and isUgcWord then
            table.insert(actionParts, {part = obj, prompt = obj:FindFirstChildOfClass("ProximityPrompt")})
            if #results < 5 then
                table.insert(results, obj.Name)
            end
        end
    end

    if #results > 0 then
        Config.LastUgcSummary = "UGC / Event Found: " .. table.concat(results, ", ")
    else
        Config.LastUgcSummary = "No direct UGC/Event object names detected in Workspace."
    end
    ugcStatus.Text = Config.LastUgcSummary
    return actionParts
end

makeToggle(pUgc, "AutoUgcEvent", function() return Config.AutoUgcEvent end, function(s) Config.AutoUgcEvent = s end)
makeButton(pUgc, "ScanUgc", function()
    scanUgcTargets()
end)

local function clearAllEsp()
    for targetObj, data in pairs(EspVisuals) do
        if data.Box then pcall(function() data.Box:Destroy() end) end
        if data.Billboard then pcall(function() data.Billboard:Destroy() end) end
        if data.Beam then pcall(function() data.Beam:Destroy() end) end
        if data.TargetAtt then pcall(function() data.TargetAtt:Destroy() end) end
        EspVisuals[targetObj] = nil
    end
end

end

local function collectEspCandidates()
    local list = {}
    local myChar = LocalPlayer.Character

    -- Players
    if Config.EspTargets.Player then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
                local hum = plr.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    table.insert(list, {
                        key = plr.Character,
                        adornee = hrp,
                        label = plr.DisplayName .. " [Player]",
                        category = "Player"
                    })
                end
            end
        end
    end

    -- NPC / Entity / Item in Workspace
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Model") and obj ~= myChar and not Players:GetPlayerFromCharacter(obj) then
            local hum = obj:FindFirstChildOfClass("Humanoid")
            local hrp = obj:FindFirstChild("HumanoidRootPart") or obj.PrimaryPart
            if hum and hrp and hum.Health > 0 then
                local nLower = string.lower(obj.Name)
                local isNpc = string.find(nLower, "npc") or string.find(nLower, "shop") or string.find(nLower, "quest") or string.find(nLower, "vendor")
                if isNpc and Config.EspTargets.NPC then
                    table.insert(list, {
                        key = obj,
                        adornee = hrp,
                        label = obj.Name .. " [NPC]",
                        category = "NPC"
                    })
                elseif not isNpc and Config.EspTargets.Entity then
                    table.insert(list, {
                        key = obj,
                        adornee = hrp,
                        label = obj.Name .. " [Entity]",
                        category = "Entity"
                    })
                end
            end
        elseif obj:IsA("Tool") and Config.EspTargets.Item then
            local handle = obj:FindFirstChild("Handle")
            if handle and handle:IsA("BasePart") then
                table.insert(list, {
                    key = obj,
                    adornee = handle,
                    label = obj.Name .. " [Item]",
                    category = "Item"
                })
            end
        end
    end

    return list
end

local function updateEspSystem()
    if not (Config.EspMaster and Config.EspSubEnabled) then
        return
    end

    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local originAtt = ensurePlayerOriginAttachment()
    if not root or not originAtt then return end

    local candidates = collectEspCandidates()
    local activeKeys = {}

    -- จำกัดจำนวน ESP ตาม MaxPerfScale (สูงสุด 9,000,000) โดยกันไม่ให้จอรกเกินไป
    local maxRendered = math.clamp(math.floor(Config.MaxPerfScale / 100000), 15, 90)
    local count = 0

    for _, item in ipairs(candidates) do
        if count >= maxRendered then break end
        if item.adornee and item.adornee.Parent then
            count += 1
            activeKeys[item.key] = true

            local entry = EspVisuals[item.key]
            if not entry then
                entry = {}
                EspVisuals[item.key] = entry
            end

            local distStuds = (item.adornee.Position - root.Position).Magnitude
            local distMeters = distStuds * STUDS_TO_METERS

            -- กล่องสี่เหลี่ยมรอบเป้าหมาย (3D Box)
            if Config.EspBox then
                if not entry.Box then
                    local box = Instance.new("SelectionBox")
                    box.Name = "MokoEspBox"
                    box.Color3 = Theme.EspBoxColor
                    box.LineThickness = 0.04
                    box.SurfaceTransparency = 0.85
                    box.SurfaceColor3 = Theme.PurplePrimary
                    box.Adornee = item.adornee
                    box.Parent = ScreenGui
                    entry.Box = box
                else
                    entry.Box.Adornee = item.adornee
                end
            elseif entry.Box then
                entry.Box:Destroy()
                entry.Box = nil
            end

            -- ระยะทางนับเป็นเมตรด้านบนกล่องสี่เหลี่ยม
            if Config.EspMetersTop then
                if not entry.Billboard then
                    local bb = Instance.new("BillboardGui")
                    bb.Name = "MokoEspMeterTop"
                    bb.Size = UDim2.new(0, 140, 0, 34)
                    bb.StudsOffset = Vector3.new(0, 3.6, 0)
                    bb.AlwaysOnTop = true
                    bb.Adornee = item.adornee
                    bb.Parent = ScreenGui

                    local txt = Instance.new("TextLabel")
                    txt.Name = "MeterLabel"
                    txt.Size = UDim2.new(1, 0, 1, 0)
                    txt.BackgroundTransparency = 1
                    txt.TextColor3 = Theme.TextMain
                    txt.TextStrokeTransparency = 0.2
                    txt.TextStrokeColor3 = Theme.PurpleDark
                    txt.Font = Enum.Font.GothamBold
                    txt.TextSize = 12
                    txt.Parent = bb

                    entry.Billboard = bb
                    entry.MeterLabel = txt
                else
                    entry.Billboard.Adornee = item.adornee
                end
                if entry.MeterLabel then
                    entry.MeterLabel.Text = string.format("%.1f m\n%s", distMeters, item.label)
                end
            elseif entry.Billboard then
                entry.Billboard:Destroy()
                entry.Billboard = nil
                entry.MeterLabel = nil
            end

            -- เส้นโยงจากตัวผู้เล่นไปยังกล่องสี่เหลี่ยม
            if Config.EspTracer then
                if not entry.TargetAtt then
                    local tAtt = Instance.new("Attachment")
                    tAtt.Name = "MokoEspTargetAtt"
                    tAtt.Parent = item.adornee
                    entry.TargetAtt = tAtt
                end
                if not entry.Beam then
                    local beam = Instance.new("Beam")
                    beam.Name = "MokoEspTracerLine"
                    beam.Attachment0 = originAtt
                    beam.Attachment1 = entry.TargetAtt
                    beam.Color = ColorSequence.new(Theme.EspLineColor, Theme.PurpleBright)
                    beam.Width0 = 0.06
                    beam.Width1 = 0.06
                    beam.FaceCamera = true
                    beam.Parent = item.adornee
                    entry.Beam = beam
                else
                    entry.Beam.Attachment0 = originAtt
                    entry.Beam.Attachment1 = entry.TargetAtt
                end
            else
                if entry.Beam then entry.Beam:Destroy() entry.Beam = nil end
                if entry.TargetAtt then entry.TargetAtt:Destroy() entry.TargetAtt = nil end
            end
        end
    end

    for key, _ in pairs(EspVisuals) do
        if not activeKeys[key] then
            local data = EspVisuals[key]
            if data.Box then pcall(function() data.Box:Destroy() end) end
            if data.Billboard then pcall(function() data.Billboard:Destroy() end) end
            if data.Beam then pcall(function() data.Beam:Destroy() end) end
            if data.TargetAtt then pcall(function() data.TargetAtt:Destroy() end) end
            EspVisuals[key] = nil
        end
    end
end

local lastAttackTime = 0
local lastUgcTick = 0
local lastEspTick = 0

trackConn(RunService.Stepped:Connect(function(now)
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local root = char and char:FindFirstChild("HumanoidRootPart")

    if hum then
        if hum.WalkSpeed ~= Config.WalkSpeed then hum.WalkSpeed = Config.WalkSpeed end
        if hum.JumpPower ~= Config.JumpPower then hum.JumpPower = Config.JumpPower end
    end

    if Config.NoClip and char then
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") then
                p.CanCollide = false
            end
        end
    end

    -- ระบบฟาร์ม / ฆ่า / โจมตี
    if (Config.AutoFarm or Config.AutoKill) and hum and root then
        local targetPart, dist = findNearestCombatTarget()
        if targetPart then
            if dist > 6 then
                hum:MoveTo(targetPart.Position)
            end
            if dist <= Config.AttackReach and (now - lastAttackTime) >= Config.HitInterval then
                lastAttackTime = now
                local tool = char:FindFirstChildOfClass("Tool")
                if tool then
                    pcall(function() tool:Activate() end)
                end
            end
        end
    end

    -- ระบบช่วยทำ UGC Event ทุกแมพ
    if Config.AutoUgcEvent and hum and root and (now - lastUgcTick) >= 1.2 then
        lastUgcTick = now
        local ugcList = scanUgcTargets()
        if #ugcList > 0 then
            local top = ugcList[1]
            local d = (top.part.Position - root.Position).Magnitude
            if d > 7 then
                hum:MoveTo(top.part.Position)
            elseif top.prompt and top.prompt.Enabled then
                pcall(function()
                    fireproximityprompt(top.prompt)
                end)
            end
        end
    end

    -- อัปเดตระบบ ESP แบบกล่อง 3D + ระยะเมตรด้านบน + เส้นโยง
    if Config.EspMaster and Config.EspSubEnabled and (now - lastEspTick) >= 0.06 then
        lastEspTick = now
        updateEspSystem()
    end
end))
