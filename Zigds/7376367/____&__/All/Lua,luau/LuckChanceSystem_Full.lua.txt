local function clampValue(v)
	local num = tonumber(v) or 0
	return math.clamp(math.floor(num + 0.5), 0, CONFIG.MaxValue)
end

local function formatNumber(n)
	n = clampValue(n)
	local str = tostring(n)
	while true do
		local k
		str, k = string.gsub(str, "^(-?%d+)(%d%d%d)", "%1,%2")
		if k == 0 then break end
	end
	return str
end

local function formatCompact(n)
	n = clampValue(n)
	if n >= 1000000 then
		return string.format("%.2fM", n / 1000000)
	elseif n >= 1000 then
		return string.format("%.0fK", n / 1000)
	end
	return tostring(n)
end

-- ฟังก์ชันซิงค์ค่าโชคลาภและโอกาสเข้ากับตัวผู้เล่นและระบบเกม
local function applyStatsToPlayer()
	local activeLuck = State.LuckEnabled and State.LuckValue or 0
	local activeChance = State.ChanceEnabled and State.ChanceValue or 0

	-- 1) บันทึกเป็น Attributes บน Instance ของ Player (เรียกใช้ได้จากทุกสคริปต์)
	player:SetAttribute("LuckBoostEnabled", State.LuckEnabled)
	player:SetAttribute("LuckBoostValue", activeLuck)
	player:SetAttribute("ChanceBoostEnabled", State.ChanceEnabled)
	player:SetAttribute("ChanceBoostValue", activeChance)

	-- 2) หากมี Folder ชื่อ leaderstats หรือ Stats ที่มีค่า Luck / Chance จะอัปเดตให้ด้วย (ฝั่ง Client Preview)
	local statsFolder = player:FindFirstChild("leaderstats") or player:FindFirstChild("Stats")
	if statsFolder then
		local luckObj = statsFolder:FindFirstChild("Luck")
		if luckObj and (luckObj:IsA("IntValue") or luckObj:IsA("NumberValue")) then
			luckObj.Value = activeLuck
		end
		local chanceObj = statsFolder:FindFirstChild("Chance")
		if chanceObj and (chanceObj:IsA("IntValue") or chanceObj:IsA("NumberValue")) then
			chanceObj.Value = activeChance
		end
	end

	-- 3) ส่งข้อมูลผ่าน RemoteEvent หากเซิร์ฟเวอร์มีรองรับ
	local remote = ReplicatedStorage:FindFirstChild(CONFIG.RemoteEventName)
	if remote and remote:IsA("RemoteEvent") then
		remote:FireServer({
			LuckEnabled = State.LuckEnabled,
			LuckValue = activeLuck,
			ChanceEnabled = State.ChanceEnabled,
			ChanceValue = activeChance,
		})
	end
end

local function tween(obj, duration, props)
	local t = TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), props)
	t:Play()
	return t
end

-- ระบบลากอิสระ (รองรับทั้ง Mouse และ Touch Screen)
local function makeDraggable(dragHandle, targetFrame, onDragStart, onDragEnd)
	local dragging = false
	local dragStart, startPos

	dragHandle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = targetFrame.Position
			if onDragStart then onDragStart() end
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			targetFrame.Position = UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset + delta.X,
				startPos.Y.Scale,
				startPos.Y.Offset + delta.Y
			)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			if dragging then
				dragging = false
				if onDragEnd then onDragEnd() end
			end
		end
	end)
end

			targetVal = clampValue(math.floor((raw + CONFIG.SliderStep / 2) / CONFIG.SliderStep) * CONFIG.SliderStep)
		end
		opts.SetValue(targetVal)
		refreshUI()
		applyStatsToPlayer()
	end

	sliderBg.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			isSliding = true
			applySliderX(input.Position.X)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if isSliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			applySliderX(input.Position.X)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			isSliding = false
		end
	end)

	refreshUI()
	return refreshUI
end
