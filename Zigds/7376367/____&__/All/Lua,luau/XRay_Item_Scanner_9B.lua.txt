-- ==========================================
-- HELPER FUNCTIONS (FORMAT NUMBER & RANGE)
-- ==========================================
local function formatNumber(num)
	num = tonumber(num) or 0
	local absNum = math.abs(num)
	if absNum >= 1e9 then
		return string.format("%.2fB", num / 1e9)
	elseif absNum >= 1e6 then
		return string.format("%.2fM", num / 1e6)
	elseif absNum >= 1e3 then
		return string.format("%.2fK", num / 1e3)
	else
		return tostring(math.floor(num))
	end
end

local function formatFullMoney(num)
	num = tonumber(num) or 0
	local formatted = tostring(math.floor(num))
	while true do
		local k
		formatted, k = string.gsub(formatted, "^(-?%d+)(%d%d%d)", "%1,%2")
		if k == 0 then break end
	end
	return "$" .. formatted .. " (" .. formatNumber(num) .. ")"
end

local function clampRange(val)
	val = tonumber(val) or Config.MinRange
	return math.clamp(math.floor(val), Config.MinRange, Config.HardCapRange)
end

local function getRootPosition()
	local character = player.Character
	if not character then return nil end
	local hrp = character:FindFirstChild("HumanoidRootPart") or character.PrimaryPart
	return hrp and hrp.Position or nil
end

-- ==========================================
-- DRAG FUNCTION (วงกลมลายน้ำ & หน้าต่างหลัก)
-- ==========================================
local function makeDraggable(guiObject)
	local dragging = false
	local dragInput, dragStart, startPos

	guiObject.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = guiObject.Position

			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	guiObject.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			local delta = input.Position - dragStart
			guiObject.Position = UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset + delta.X,
				startPos.Y.Scale,
				startPos.Y.Offset + delta.Y
			)
		end
	end)
end

local activeESP = {}

local function getTargetData(instance)
	local name = instance.Name
	local dbEntry = ItemDatabase[name]

	-- อ่านค่าจาก Attribute หรือ Value ข้างในถ้ามี
	local thName = instance:GetAttribute("NameTH") or (dbEntry and dbEntry.th) or name
	local enName = instance:GetAttribute("NameEN") or (dbEntry and dbEntry.en) or name
	local value = instance:GetAttribute("Value") or instance:GetAttribute("Price") or (dbEntry and dbEntry.value) or 0
	local rarity = instance:GetAttribute("Rarity") or (dbEntry and dbEntry.rarity) or "Common"

	local isUnderground = (instance:GetAttribute("Underground") == true)
		or (instance:GetAttribute("IsOre") == true)
		or (instance:GetAttribute("IsBlock") == true)
		or (dbEntry ~= nil and string.find(string.lower(enName), "crystal") ~= nil)

	local isItem = (instance:GetAttribute("IsItem") == true)
		or instance:IsA("Tool")
		or (dbEntry ~= nil and not isUnderground)

	-- ตรวจจับอัตโนมัติจากชื่อหรือตำแหน่งใต้ดิน
	if not isItem and not isUnderground then
		local basePart = instance:IsA("BasePart") and instance or instance:FindFirstChildWhichIsA("BasePart", true)
		if basePart and basePart.Position.Y < -5 then
			isUnderground = true
		elseif string.find(string.lower(name), "item") or string.find(string.lower(name), "drop") or string.find(string.lower(name), "loot") then
			isItem = true
		end
	end

	if not isItem and not isUnderground then
		return nil
	end

	return {
		thName = tostring(thName),
		enName = tostring(enName),
		value = tonumber(value) or 0,
		rarity = tostring(rarity),
		isItem = isItem,
		isUnderground = isUnderground,
		color = (dbEntry and dbEntry.color) or RarityColors[rarity] or Color3.fromRGB(168, 85, 247),
	}
end

local function clearAllESP()
	for target, entry in pairs(activeESP) do
		if entry.highlight then entry.highlight:Destroy() end
		if entry.billboard then entry.billboard:Destroy() end
		activeESP[target] = nil
	end
end

local function scanWorld()
	local rootPos = getRootPosition()
	if not Config.Enabled or not rootPos then
		clearAllESP()
		statusLabel.Text = "สถานะ: ปิดใช้งาน หรือรอตัวละครผู้เล่น"
		return
	end

	local seenTargets = {}
	local foundCount = 0

	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj:IsA("BasePart") or obj:IsA("Model") or obj:IsA("Tool") then
			local data = getTargetData(obj)
			if data then
				if (data.isItem and Config.ScanItems) or (data.isUnderground and Config.ScanUnderground) then
					local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart", true)
					if part then
						local dist = (part.Position - rootPos).Magnitude
						if dist <= Config.MaxRange then
							seenTargets[obj] = true
							foundCount += 1
							createOrUpdateESP(obj, data, dist)
						end
					end
				end
			end
		end
	end

	for target, entry in pairs(activeESP) do
		if not seenTargets[target] or not target.Parent then
			if entry.highlight then entry.highlight:Destroy() end
			if entry.billboard then entry.billboard:Destroy() end
			activeESP[target] = nil
		end
	end

	statusLabel.Text = string.format(
		"ระยะ: %s / 9B | พบทั้งหมด: %d ชิ้น\nไอเทม: %s | ใต้ดิน/บล็อก: %s",
		formatNumber(Config.MaxRange),
		foundCount,
		Config.ScanItems and "เปิด" or "ปิด",
		Config.ScanUnderground and "เปิด" or "ปิด"
	)
end
