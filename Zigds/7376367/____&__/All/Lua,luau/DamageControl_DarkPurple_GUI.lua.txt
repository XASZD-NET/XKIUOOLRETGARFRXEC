-- ============================================================================
local function makeDraggable(guiObject)
	local dragging = false
	local dragInput, dragStart, startPos

	guiObject.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = guiObject.Position

			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	guiObject.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			local delta = input.Position - dragStart
			guiObject.Position = UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset + delta.X,
				startPos.Y.Scale,
				startPos.Y.Offset + delta.Y
			)
		end
	end)
end

-- ฟังก์ชันจัดรูปแบบตัวเลขให้มีลูกน้ำ (เช่น 9,000,000)
local function formatNumber(n)
	local formatted = tostring(math.floor(n))
	while true do
		local k
		formatted, k = string.gsub(formatted, "^(-?%d+)(%d%d%d)", "%1,%2")
		if k == 0 then break end
	end
	return formatted
end

-- ระบบอัปเดตค่าและเชื่อมต่อการทำงาน
-- ============================================================================
local function setDamageValue(newValue)
	newValue = math.clamp(math.floor(newValue), 1, MAX_DAMAGE)
	currentDamage = newValue
	
	local ratio = newValue / MAX_DAMAGE
	sliderFill.Size = UDim2.new(ratio, 0, 1, 0)
	sliderKnob.Position = UDim2.new(ratio, 0, 0.5, 0)
	
	valueDisplay.Text = "พลังโจมตี: " .. formatNumber(newValue)
	inputBox.Text = tostring(newValue)
end

-- ระบบลากแถบเลื่อน (Slider)
local sliding = false
local function updateSliderFromInput(input)
	local posX = math.clamp(input.Position.X - sliderBg.AbsolutePosition.X, 0, sliderBg.AbsoluteSize.X)
	local ratio = posX / sliderBg.AbsoluteSize.X
	local calculatedValue = math.max(1, math.floor(ratio * MAX_DAMAGE))
	setDamageValue(calculatedValue)
end

sliderBg.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		sliding = true
		updateSliderFromInput(input)
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		sliding = false
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		updateSliderFromInput(input)
	end
end)

-- เมื่อพิมพ์ตัวเลขในกล่องข้อความเสร็จ
inputBox.FocusLost:Connect(function()
	local num = tonumber(inputBox.Text)
	if num then
		setDamageValue(num)
	else
		inputBox.Text = tostring(currentDamage)
	end
end)

-- ปุ่มกดยืนยันส่งค่าดาเมจ
applyBtn.MouseButton1Click:Connect(function()
	-- 1. บันทึกค่าไว้ใน Attribute ของผู้เล่น (สคริปต์อาวุธ/สกิลดึงไปใช้ได้ทันที)
	player:SetAttribute("CustomDamage", currentDamage)
	
	-- 2. ค้นหาและปรับค่า Damage ใน Tool ที่ผู้เล่นถืออยู่หรืออยู่ในกระเป๋า (หากมี Value ชื่อ Damage)
	local function updateTools(container)
		if not container then return end
		for _, item in ipairs(container:GetChildren()) do
			if item:IsA("Tool") then
				item:SetAttribute("Damage", currentDamage)
				local dmgVal = item:FindFirstChild("Damage") or item:FindFirstChild("Dmg") or item:FindFirstChild("Power")
				if dmgVal and (dmgVal:IsA("NumberValue") or dmgVal:IsA("IntValue")) then
					dmgVal.Value = currentDamage
				end
			end
		end
	end
	updateTools(player.Character)
	updateTools(player:FindFirstChild("Backpack"))

	-- 3. ส่งค่าไปยัง Server ผ่าน RemoteEvent (หากสร้างไว้)
	if damageEvent then
		damageEvent:FireServer(currentDamage)
	end
	
	-- เอฟเฟกต์แจ้งเตือนบนปุ่ม
	applyBtn.Text = "✓ บันทึกแล้ว!"
	applyBtn.BackgroundColor3 = Color3.fromRGB(90, 180, 110)
	task.wait(1)
	applyBtn.Text = "ใช้งานค่านี้"
	applyBtn.BackgroundColor3 = Color3.fromRGB(125, 45, 220)
end)
